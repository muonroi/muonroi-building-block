## AngleSharp CSS Inspection API [ASSUMED — verify on first build]

```csharp
// Pass 1 — Stylesheet AST walk
using AngleSharp.Css.Dom;

foreach (var sheet in document.StyleSheets.OfType<ICssStyleSheet>())
{
    foreach (var rule in sheet.Rules)
    {
        // @import with external URI
        if (rule is ICssImportRule importRule && IsExternalUri(importRule.Href))
        {
            violations.Add(new PolicyViolation(
                "forbidden.import.external", $"External @import is not allowed: {importRule.Href}",
                CssSelector: "@import", PropertyName: "@import",
                RejectedValue: importRule.Href, SuggestedAlternative: "Inline the stylesheet"));
        }

        // @keyframes (animation)
        if (rule is ICssKeyframesRule keyframesRule)
        {
            violations.Add(new PolicyViolation(
                "forbidden.css-animation", $"CSS animations (@keyframes) are not supported: {keyframesRule.Name}",
                CssSelector: $"@keyframes {keyframesRule.Name}", PropertyName: "animation",
                RejectedValue: $"@keyframes {keyframesRule.Name}", SuggestedAlternative: "Remove animation properties"));
        }

        // transition in style rule declarations
        if (rule is ICssStyleRule styleRule)
        {
            var transition = styleRule.Style?.GetPropertyValue("transition");
            if (!string.IsNullOrEmpty(transition))
            {
                violations.Add(new PolicyViolation(
                    "forbidden.css-transition", "CSS transitions are not supported",
                    CssSelector: styleRule.SelectorText, PropertyName: "transition",
                    RejectedValue: transition, SuggestedAlternative: "Remove transition properties"));
            }
        }
    }
}

// Pass 2 — Computed style per element
if (document.DefaultView is IWindowCss windowCss)
{
    foreach (var element in document.All.OfType<IElement>())
    {
        var style = windowCss.GetComputedStyle(element);
        if (style is null) continue;

        var display = style.GetPropertyValue("display") ?? string.Empty;
        var floatVal = style.GetPropertyValue("float") ?? string.Empty;
        var position = style.GetPropertyValue("position") ?? string.Empty;
        string selector = element.GetSelector() ?? element.LocalName;

        if (display is "flex" or "inline-flex")
            violations.Add(ViolationFor("forbidden.display.flex", "display", display, selector, "display:block"));
        if (display is "grid" or "inline-grid")
            violations.Add(ViolationFor("forbidden.display.grid", "display", display, selector, "display:table"));
        if (floatVal is "left" or "right")
            violations.Add(ViolationFor("forbidden.float", "float", floatVal, selector, "Use display:table layout"));
        if (position is "absolute")
            violations.Add(ViolationFor("forbidden.position.absolute", "position", position, selector, "position:static"));
        if (position is "fixed")
            violations.Add(ViolationFor("forbidden.position.fixed", "position", position, selector, "position:static"));
        if (position is "sticky")
            violations.Add(ViolationFor("forbidden.position.sticky", "position", position, selector, "position:static"));
    }
}
```

## GOV-03 SignedPdfCssPolicyDecorator

The decorator wraps any `IPdfCssPolicy` and adds signature verification when `PdfConfigs.RequirePolicySignature = true`.

**Design note (Pitfall 5 from RESEARCH.md)**: `PolicyVerifier.Verify(LicensePolicy)` takes `LicensePolicy`, not `PdfConfigs`. The decorator must NOT create a `LicensePolicy` coupling. Instead, implement the decorator with a `Func<bool>` signature verifier injected in the constructor — Phase 6 DI will wire it. This keeps the decorator testable without requiring the full license infrastructure.

```csharp
public sealed class SignedPdfCssPolicyDecorator : IPdfCssPolicy
{
    private readonly IPdfCssPolicy _inner;
    private readonly PdfConfigs _configs;
    private readonly Func<bool> _signatureVerifier;   // injected; Phase 6 wires PolicyVerifier

    public SignedPdfCssPolicyDecorator(
        IPdfCssPolicy inner,
        PdfConfigs configs,
        Func<bool> signatureVerifier)
    {
        _inner = inner;
        _configs = configs;
        _signatureVerifier = signatureVerifier;
    }

    public string Id => _inner.Id;
    public PdfPolicyLimits Limits => _inner.Limits;

    public async ValueTask<PolicyValidationResult> ValidateAsync(
        IPdfDocumentContext documentContext, CancellationToken ct = default)
    {
        if (_configs.RequirePolicySignature && !_signatureVerifier())
        {
            throw new PdfPolicyException(new[]
            {
                new PolicyViolation(
                    "gov.policy.signature-invalid",
                    "Policy configuration signature is invalid or missing.",
                    PolicySeverity.Error,
                    PropertyName: "RequirePolicySignature",
                    RejectedValue: "unsigned",
                    SuggestedAlternative: "Sign the policy config with the Muonroi policy signing tool")
            });
        }
        return await _inner.ValidateAsync(documentContext, ct);
    }
}
```
