## FakeStyledNode Usage

```csharp
// Block with margin-top: 20px

// Block with overflow:hidden (BFC root)

    ["margin-top"] = "10px" });

// Inline text
var span = new FakeStyledNode("span", new() { ["display"] = "inline", ["vertical-align"] = "top" });

// Display none

// Table with border-spacing

// SC1: Adjacent margin collapsing
[Fact]
public void AdjacentBlocks_MarginCollapsesTo_Maximum()
{
    // Arrange: two adjacent blocks, first has margin-bottom:20px, second has margin-top:30px
    var parent = new FakeStyledNode("div", new() { ["display"] = "block" });
    var child1 = new FakeStyledNode("p", new() { ["display"] = "block", ["margin-bottom"] = "20px" });
    child1.TextContent = "First";
    var child2 = new FakeStyledNode("p", new() { ["display"] = "block", ["margin-top"] = "30px" });
    child2.TextContent = "Second";
    parent.ChildList.Add(child1);
    parent.ChildList.Add(child2);

    var builder = new BoxTreeBuilder();
    var rootBox = builder.Build(parent) as BlockBox;
    var engine = new BlockLayoutEngine(new InlineLayoutEngine());
    var context = MakeContext(availableWidth: 400f, pageHeight: 800f);
    var elements = new List<PositionedElement>();
    engine.Layout(rootBox!, context, elements, pageIndex: 0);

    // Gap between child1 bottom and child2 top should be max(20, 30) = 30pt * PxToPt = 22.5pt
    // NOT 50pt (which would be the sum)
    float child1Bottom = elements.First(e => e.Source == rootBox!.Children[0])
        .Position.Y + elements.First(e => e.Source == rootBox!.Children[0]).Position.Height;
    float child2Top = elements.First(e => e.Source == rootBox!.Children[1]).Position.Y;
    float gap = child2Top - child1Bottom;
    gap.Should().BeApproximately(30f * 0.75f, precision: 1f); // 30px * PxToPt
}

// LayoutEngine.Layout returns non-empty IPositionedPageList
[Fact]
public void LayoutEngine_Layout_ReturnsNonEmptyPageList()
{
    var htmlRoot = new FakeStyledNode("html", new() { ["display"] = "block" });
    var body = new FakeStyledNode("body", new() { ["display"] = "block" });
    var p = new FakeStyledNode("p", new() { ["display"] = "block" });
    p.TextContent = "Hello PDF";
    body.ChildList.Add(p);
    htmlRoot.ChildList.Add(body);

    var doc = new FakeStyledDocument(htmlRoot, pageRule: null);
    var engine = new LayoutEngine();
    var result = engine.Layout(doc, new PdfRenderOptions(), PdfConfigs.Limits.Default);

    result.Should().NotBeNull();
    ((PositionedPageList)result).PageCount.Should().BeGreaterThan(0);
}
```

        AvailableWidth = availableWidth,
