public interface IStyledNode
{
    string LocalName { get; }
    string? TextContent { get; }

    string? GetAttribute(string name);
    bool IsElement { get; }

public interface IComputedStyle
{

public interface IPageRule
{
    PdfMargins Margins { get; }

}
