## AngleSharpStyledNode Implementation

```csharp
// src/Muonroi.Pdf.Governance/Cascade/AngleSharpStyledNode.cs
internal sealed class AngleSharpStyledNode : IStyledNode
{
    private readonly INode _node;
    private readonly IWindow? _window;

    internal AngleSharpStyledNode(INode node, IWindow? window)
    {
        _node = node;
        _window = window;
    }

    public string LocalName => (_node as IElement)?.LocalName ?? "#text";
    public string? TextContent => _node.TextContent;
    public bool IsElement => _node is IElement;
    public bool IsText => _node.NodeType == NodeType.Text;
    public string? GetAttribute(string name) => (_node as IElement)?.GetAttribute(name);

    public IComputedStyle Style
    {
        get
        {
            if (_node is IElement element && _window is not null)
                return new AngleSharpComputedStyle(_window.ComputedStyle(element));
            return AngleSharpComputedStyle.Empty;
        }
    }

    public IReadOnlyList<IStyledNode> Children
    {
        get
        {
            var result = new List<IStyledNode>(_node.ChildNodes.Length);
            foreach (INode child in _node.ChildNodes)
            {
                if (child is IElement || child.NodeType == NodeType.Text)
                    result.Add(new AngleSharpStyledNode(child, _window));
            }
            return result;
        }
    }
}
```

## AngleSharpPageRule Implementation

`@page` rules are CSS at-rules. Access via `document.StyleSheets` then cast each rule to `ICssPageRule` (AngleSharp.Css provides this). If no @page rule is found, return null.

```csharp
// src/Muonroi.Pdf.Governance/Cascade/AngleSharpPageRule.cs
internal sealed class AngleSharpPageRule : IPageRule
{
    public PdfMargins Margins { get; }
    public string? TopMarginBoxHtml { get; }
    public string? BottomMarginBoxHtml { get; }
    public string? Size { get; }

    internal AngleSharpPageRule(PdfMargins margins, string? topHtml, string? bottomHtml, string? size)
    {
        Margins = margins;
        TopMarginBoxHtml = topHtml;
        BottomMarginBoxHtml = bottomHtml;
        Size = size;
    }

    internal static IPageRule? TryExtract(IDocument document)
    {
        foreach (IStyleSheet sheet in document.StyleSheets)
        {
            if (sheet is not ICssStyleSheet cssSheet) continue;
            foreach (ICssRule rule in cssSheet.Rules)
            {
                if (rule is not ICssPageRule pageRule) continue;
                var margins = ParsePageMargins(pageRule.Style);
                var size = pageRule.Style.GetPropertyValue("size");
                return new AngleSharpPageRule(margins, topHtml: null, bottomHtml: null,
                    string.IsNullOrEmpty(size) ? null : size);
            }
        }
        return null;
    }

    private static PdfMargins ParsePageMargins(ICssStyleDeclaration style)
    {
        static float ParseMm(string? val)
        {
            if (string.IsNullOrEmpty(val)) return 10f;
            if (val.EndsWith("mm", StringComparison.OrdinalIgnoreCase) &&
                float.TryParse(val[..^2], out float mm)) return mm;
            if (val.EndsWith("px", StringComparison.OrdinalIgnoreCase) &&
                float.TryParse(val[..^2], out float px)) return px * 0.264583f;
            return 10f;
        }

        string? all = style.GetPropertyValue("margin");
        if (!string.IsNullOrEmpty(all))
        {
            float v = ParseMm(all.Split(' ')[0]);
            return new PdfMargins { Top = v, Right = v, Bottom = v, Left = v };
        }

        return new PdfMargins
        {
            Top = ParseMm(style.GetPropertyValue("margin-top")),
            Right = ParseMm(style.GetPropertyValue("margin-right")),
            Bottom = ParseMm(style.GetPropertyValue("margin-bottom")),
            Left = ParseMm(style.GetPropertyValue("margin-left"))
        };
    }
}
```
